Finally, let us conduct several experiments. 

from random import randint, seed
from statistics import mean


seed(17)
number_of_days = 365
experiments = []

for _ in range(10 ** 3):
    birthdays = set()
    while True:
        date = randint(1, number_of_days)
        if date not in birthdays:
            birthdays.add(date)
        else:
            experiments.append(len(birthdays) + 1)
            break

print(mean(experiments))
