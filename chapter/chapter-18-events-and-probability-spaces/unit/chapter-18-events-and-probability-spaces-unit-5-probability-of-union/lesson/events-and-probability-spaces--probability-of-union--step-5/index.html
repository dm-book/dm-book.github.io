from itertools import permutations

for n in range(2, 10):
    total, good = 0, 0
    for p in permutations(range(n)):
        total += 1
        if all(p[i] != i for i in range(n)):
            good += 1
    print(f'{n} {good/total:.7f} {total/good:.7f}')
